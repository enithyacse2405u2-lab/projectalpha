1. Data Encryption and Decryption

#include <stdio.h>

int main()
{
    char text[100];
    int key, i;

    printf("Enter the text: ");
    fgets(text, sizeof(text), stdin);

    printf("Enter the key: ");
    scanf("%d", &key);

    // Encryption
    for(i = 0; text[i] != '\0'; i++)
    {
        if(text[i] != ' ' && text[i] != '\n')
            text[i] = text[i] + key;
    }

    printf("Encrypted data: %s", text);

    // Decryption
    for(i = 0; text[i] != '\0'; i++)
    {
        if(text[i] != ' ' && text[i] != '\n')
            text[i] = text[i] - key;
    }

    printf("Decrypted data: %s", text);

    return 0;
}

2. Leaky Bucket Algorithm

#include <stdio.h>

int main()
{
    int bucket_size, output_rate, n, packet;
    int storage = 0;
    int i;

    printf("Enter bucket size: ");
    scanf("%d", &bucket_size);

    printf("Enter output rate: ");
    scanf("%d", &output_rate);

    printf("Enter number of packets: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        printf("\\nEnter packet size: ");
        scanf("%d", &packet);

        if(packet <= bucket_size - storage)
        {
            storage = storage + packet;
            printf("Packet accepted\\n");
        }
        else
        {
            printf("Packet dropped\\n");
        }

        if(storage >= output_rate)
        {
            storage = storage - output_rate;
            printf("Packets sent: %d\\n", output_rate);
        }
        else
        {
            printf("Packets sent: %d\\n", storage);
            storage = 0;
        }

        printf("Remaining packets: %d\\n", storage);
    }

    return 0;
}

3. Link State Routing Algorithm

#include <stdio.h>

#define INF 999

int main()
{
    int n, cost[10][10], dist[10];
    int visited[10] = {0};
    int i, j, src, min, u;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter cost matrix (999 for no connection):\\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < n; j++)
            scanf("%d", &cost[i][j]);
    }

    printf("Enter source node (0 to %d): ", n - 1);
    scanf("%d", &src);

    for(i = 0; i < n; i++)
        dist[i] = cost[src][i];

    dist[src] = 0;
    visited[src] = 1;

    for(i = 1; i < n; i++)
    {
        min = INF;
        u = -1;

        for(j = 0; j < n; j++)
        {
            if(!visited[j] && dist[j] < min)
            {
                min = dist[j];
                u = j;
            }
        }

        if(u == -1)
            break;

        visited[u] = 1;

        for(j = 0; j < n; j++)
        {
            if(!visited[j] && cost[u][j] != INF &&
               dist[u] + cost[u][j] < dist[j])
                dist[j] = dist[u] + cost[u][j];
        }
    }

    printf("\\nShortest Path from Node %d:\\n", src);

    for(i = 0; i < n; i++)
    {
        if(dist[i] == INF)
            printf("Node %d : Unreachable\\n", i);
        else
            printf("Node %d : %d\\n", i, dist[i]);
    }

    return 0;
}

4. Distance Vector Routing Algorithm

#include <stdio.h>

#define INF 999

int main()
{
    int n, cost[10][10], dist[10][10];
    int i, j, k;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter cost matrix (999 for no connection):\\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < n; j++)
        {
            scanf("%d", &cost[i][j]);
            dist[i][j] = cost[i][j];
        }
    }

    for(k = 0; k < n; k++)
    {
        for(i = 0; i < n; i++)
        {
            for(j = 0; j < n; j++)
            {
                if(dist[i][k] != INF && dist[k][j] != INF &&
                   dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];
            }
        }
    }

    printf("\\nRouting Tables:\\n");

    for(i = 0; i < n; i++)
    {
        printf("\\nRouter %d:\\n", i);
        printf("Destination\\tDistance\\n");

        for(j = 0; j < n; j++)
        {
            if(dist[i][j] == INF)
                printf("%d\\t\\tUnreachable\\n", j);
            else
                printf("%d\\t\\t%d\\n", j, dist[i][j]);
        }
    }

    return 0;
}

5. Divide a Network into N Subnetworks

#include <stdio.h>

int main()
{
    int a, b, c, n, bits = 0;
    int prefix, hosts, i, block;

    printf("Enter Network Address (A.B.C): ");
    scanf("%d.%d.%d", &a, &b, &c);

    printf("Enter Number of Subnets: ");
    scanf("%d", &n);

    if(a < 0 || a > 255 || b < 0 || b > 255 ||
       c < 0 || c > 255 || n < 1 || n > 64)
    {
        printf("Invalid Input");
        return 0;
    }

    while((1 << bits) < n)
        bits++;

    prefix = 24 + bits;
    block = 256 / (1 << bits);
    hosts = block - 2;

    printf("\\nSubnet Mask: 255.255.255.%d", 256 - block);
    printf("\\nPrefix Length: /%d", prefix);
    printf("\\nHosts per Subnet: %d\\n", hosts);

    for(i = 0; i < n; i++)
    {
        int start = i * block;

        printf("\\nSubnet %d", i + 1);
        printf("\\nNetwork Address: %d.%d.%d.%d", a, b, c, start);
        printf("\\nFirst Host: %d.%d.%d.%d", a, b, c, start + 1);
        printf("\\nLast Host: %d.%d.%d.%d", a, b, c, start + block - 2);
        printf("\\nBroadcast Address: %d.%d.%d.%d\\n",
               a, b, c, start + block - 1);
    }

    return 0;
}

6. Dijkstra's Shortest Path Algorithm

    {

            scanf("%d", &cost[i][j]);
    }

    printf("Enter source vertex (0 to %d): ", n - 1);
    scanf("%d", &src);

    for(i = 0; i < n; i++)
        dist[i] = cost[src][i];

    dist[src] = 0;
    visited[src] = 1;

    for(i = 1; i < n; i++)
    {
        min = INF;
        u = -1;

        for(j = 0; j < n; j++)
        {
            if(!visited[j] && dist[j] < min)
            {
                min = dist[j];
                u = j;
            }
        }

        if(u == -1)
            break;

        visited[u] = 1;

        for(j = 0; j < n; j++)
        {
            if(!visited[j] && cost[u][j] != INF &&
               dist[u] + cost[u][j] < dist[j])

    printf("\\nShortest distances from vertex %d:\\n", src);

7. Classless Addressing (CIDR)

#include <stdio.h>

int main()
{
    unsigned int a, b, c, d, mask;
    int prefix;

    printf("Enter IP Address: ");
    scanf("%u.%u.%u.%u", &a, &b, &c, &d);

    printf("Enter Prefix Length: ");
    scanf("%d", &prefix);

    if(a > 255 || b > 255 || c > 255 || d > 255 ||
       prefix < 0 || prefix > 32)
    {
        printf("Invalid Input");
        return 0;
    }

    if(prefix == 0)
        mask = 0;
    else
        mask = 0xFFFFFFFFu << (32 - prefix);

    printf("\\nCIDR Address: %u.%u.%u.%u/%d", a, b, c, d, prefix);

    printf("\\nSubnet Mask: %u.%u.%u.%u",
           (mask >> 24) & 255,
           (mask >> 16) & 255,
           (mask >> 8) & 255,
           mask & 255);

    return 0;
}

8. Classful Addressing

#include <stdio.h>

int main()
{
    int a, b, c, d;

    printf("Enter IP Address: ");
    scanf("%d.%d.%d.%d", &a, &b, &c, &d);

    if(a < 0 || a > 255 || b < 0 || b > 255 ||
       c < 0 || c > 255 || d < 0 || d > 255)
    {
        printf("Invalid IP Address");
    }
    else if(a >= 1 && a <= 126)
    {
        printf("Class A\\nSubnet Mask: 255.0.0.0");
    }
    else if(a == 127)
    {
        printf("Class A - Loopback Address");
    }
    else if(a >= 128 && a <= 191)
    {
        printf("Class B\\nSubnet Mask: 255.255.0.0");
    }
    else if(a >= 192 && a <= 223)
    {
        printf("Class C\\nSubnet Mask: 255.255.255.0");
    }
    else if(a >= 224 && a <= 239)
    {
        printf("Class D - Multicast");
    }
    else if(a >= 240 && a <= 255)
    {
        printf("Class E - Reserved");
    }
    else
    {
        printf("Reserved IP Address");
    }

    return 0;
}

9. Sliding Window Protocol using Go-Back-N

#include <stdio.h>

int main()
{
    int n, w, i, j, lost;

    printf("Enter number of frames: ");
    scanf("%d", &n);

    printf("Enter window size: ");
    scanf("%d", &w);

    if(n <= 0 || w <= 0)
    {
        printf("Invalid Input");
        return 0;
    }

    i = 1;

    while(i <= n)
    {
        printf("\\nSending frames: ");

        for(j = i; j < i + w && j <= n; j++)
            printf("%d ", j);

        printf("\\nEnter lost frame (0 for no loss): ");
        scanf("%d", &lost);

        if(lost >= i && lost < j)
        {
            printf("Frame %d lost!\\n", lost);
            printf("Go-Back-N: Resending from frame %d\\n", lost);
            i = lost;
        }
        else
        {
            printf("All frames acknowledged.\\n");
            i = j;
        }
    }

    printf("\\nAll frames transmitted successfully!");

    return 0;
}
